  \item \textbf{Sturm--Liouville Instanton Stability:} We integrate the on-shell Euclidean actions for geodesic neck pinch collapse in JT and CGHS models ($I_{\mathrm{inst}} = \frac{3\pi^2}{8 G_N Q}$). \section{Introduction and the Gravitational Factorization Enigma}
\label{sec:introduction}

In the semiclassical path integral approach to quantum gravity, summing over smooth spacetime geometries with fixed asymptotic boundaries inevitably requires the inclusion of wormhole topologies that connect otherwise disjoint universe asymptotic boundaries. In two-dimensional Jackiw--Teitelboim (JT) gravity coupled to bulk matter fields, following the fiber bundle and singular geometric foundations established in Purist Monograph Vol.~I~\cite{PuristVol1Monograph2026}, the path integral with $n$ disconnected boundary components of renormalized lengths $\beta_1, \dots, \beta_n$ takes the form:
\begin{equation}
  \mathcal{Z}(\beta_1, \dots, \beta_n) = \int \frac{\mathcal{D}g\,\mathcal{D}\Phi}{\mathrm{Vol}(\mathrm{Diff})} \, \exp\left( - I_{\mathrm{JT}}[g, \Phi] - I_{\mathrm{matter}}[g, \psi] \right).
  \label{eq:gravitational_path_integral}
\end{equation}
The Euclidean JT gravity action on a two-dimensional manifold $\mathcal{M}$ with boundary $\partial\mathcal{M}$ is defined by:
\begin{equation}
  I_{\mathrm{JT}}[g, \Phi] = - S_0 \, \chi(\mathcal{M}) - \frac{1}{2} \int_{\mathcal{M}} \diff^2 x \sqrt{g} \, \Phi \, (R + 2) - \int_{\partial\mathcal{M}} \diff \tau \sqrt{\gamma} \, \Phi_b \, (K - 1),
  \label{eq:jt_action}
\end{equation}
where $\chi(\mathcal{M}) = 2 - 2g - n$ is the topological Euler characteristic, $\Phi$ is the real scalar dilaton field, and $K$ is the extrinsic curvature of the boundary $\partial\mathcal{M}$ with induced metric $\gamma_{\tau\tau} = 1/\epsilon^2$.

\subsection{The Factorization Paradox}
Integrating out the bulk dilaton $\Phi$ along a contour parallel to the imaginary axis imposes the strict delta-functional constraint:
\begin{equation}
  R(x) + 2 = 0, \quad \forall x \in \mathcal{M},
\end{equation}
pinning the geometry to surfaces of constant negative curvature $\mathcal{R} = -2$ (hyperbolic 2-manifolds $\mathbb{H}^2 / \Gamma$).

For two disconnected circular boundaries of lengths $\beta_1$ and $\beta_2$, the sum over geometries includes both the disconnected product of two hyperbolic disks $\mathcal{M}_{\mathrm{disk}} \sqcup \mathcal{M}_{\mathrm{disk}}$ and connected Euclidean wormholes (cylinders / double trumpets $\mathcal{M}_{\mathrm{cylinder}}$):
\begin{equation}
  \langle Z(\beta_1) Z(\beta_2) \rangle = Z_{\mathrm{disk}}(\beta_1) Z_{\mathrm{disk}}(\beta_2) + Z_{\mathrm{cylinder}}(\beta_1, \beta_2) + \dots
  \label{eq:wormhole_connected_correlator}
\end{equation}
Because the cylinder amplitude is strictly positive:
\begin{equation}
  Z_{\mathrm{cylinder}}(\beta_1, \beta_2) = \int_0^\infty b \, \diff b \, Z_{\mathrm{trumpet}}(\beta_1, b) \, Z_{\mathrm{trumpet}}(\beta_2, b) = \frac{\sqrt{\beta_1 \beta_2}}{2\pi(\beta_1 + \beta_2)} > 0,
  \label{eq:double_trumpet_integral}
\end{equation}
the connected correlator does not vanish:
\begin{equation}
  \langle Z(\beta_1) Z(\beta_2) \rangle_{\mathrm{conn}} = \langle Z(\beta_1) Z(\beta_2) \rangle - \langle Z(\beta_1) \rangle \langle Z(\beta_2) \rangle = Z_{\mathrm{cylinder}}(\beta_1, \beta_2) \neq 0.
\end{equation}

For a holographic boundary quantum mechanical system with a definite Hamiltonian $H$, the partition function is the trace $Z(\beta) = \Tr(\mathrm{e}^{-\beta H}) \in \mathbb{C}$. For any single quantum system, the product of traces factorizes identically:
\begin{equation}
  Z(\beta_1) Z(\beta_2) \equiv \Tr(\mathrm{e}^{-\beta_1 H}) \cdot \Tr(\mathrm{e}^{-\beta_2 H}).
\end{equation}
The non-vanishing connected correlator $\langle Z(\beta_1) Z(\beta_2) \rangle_{\mathrm{conn}} \neq 0$ is the celebrated \emph{Factorization Paradox}. While Saad, Shenker, and Stanford~\cite{SaadShenkerStanford2019} proved that this connected correlator precisely matches an ensemble average over random matrix Hamiltonians $\overline{Z(\beta_1) Z(\beta_2)}$, standard holographic duality asserts that quantum gravity in the bulk must be dual to a single, fixed unitary quantum theory, not a random ensemble.

\section{The Third-Quantized Hilbert Space \texorpdfstring{$\mathcal{H}_{\mathrm{BU}}$}{H BU} and \texorpdfstring{$\alpha$}{alpha}-Vacua}

\begin{definition}[Baby Universe Fock Space $\Hbu$]
Let $\HH \in \Hbu$ denote the Hartle--Hawking no-boundary state, representing the universe with zero disconnected closed spatial components. Disconnected baby universes are created by boundary insertion operators $\Zop(\beta)$:
\begin{equation}
  \Zop(\beta) = Z_{\mathrm{disk}}(\beta) + \sqrt{\lambda} \, (a_\beta^\dagger + a_\beta),
  \label{eq:boundary_creation_operator}
\end{equation}
where $a_\beta, a_\beta^\dagger$ are third-quantized creation and annihilation operators on $\Hbu$ satisfying canonical commutation relations:
\begin{equation}
  [a_\beta, a_{\beta'}^\dagger] = \Delta(\beta, \beta'), \quad [a_\beta, a_{\beta'}] = 0.
\end{equation}
\end{definition}

\begin{theorem}[Commutativity and Abelian Von Neumann Algebra]
\label{thm:abelian_algebra}
For any two boundary lengths $\beta_1, \beta_2 \in \R^+$, the boundary-creating operators $\Zop(\beta)$ are self-adjoint and commute on a common dense domain $\mathcal{D} \subset \Hbu$:
\begin{equation}
  [\Zop(\beta_1), \Zop(\beta_2)] = 0.
\end{equation}
Their bounded unitary Weyl flows $U_\beta(t) = \exp(i t \Zop(\beta))$ generate an abelian von Neumann algebra $\mathfrak{W} = \{U_\beta(t) : t \in \R, \beta > 0\}''$ on $\Hbu$.
\end{theorem}
\begin{proof}
In the Euclidean path integral, exchanging the ordering of boundary insertions corresponds to a diffeomorphism of the asymptotic boundary curves. Since the gravitational action $I_{\mathrm{JT}}$ and boundary counterterms depend only on the intrinsic Riemannian geometry and are invariant under boundary relabeling, the operators act symmetrically:
\begin{equation}
  \Zop(\beta_1) \Zop(\beta_2) \HH = \Zop(\beta_2) \Zop(\beta_1) \HH.
\end{equation}
Because the Hartle--Hawking state $\HH$ is cyclic for $\mathfrak{W}$, commutativity extends to all dense states in $\Hbu$, proving that $[\Zop(\beta_1), \Zop(\beta_2)] = 0$ on $\mathcal{D}$ and the unitary generators commute $[U_{\beta_1}(s), U_{\beta_2}(t)] = 0$ for all $s, t \in \R$.
\end{proof}

\begin{theorem}[Exact $\alpha$-Vacua Factorization Theorem]
\label{thm:exact_factorization}
For any designated boundary length $\beta$ (or within the topological single-mode sector), by the Spectral Decomposition Theorem for commuting unbounded self-adjoint operators, the abelian von Neumann algebra $\mathfrak{W}$ admits an exact spectral resolution. The baby universe Hilbert space $\Hbu$ decomposes into an orthogonal direct integral over the real spectrum $\R$:
\begin{equation}
  \Hbu = \int_{\R}^{\oplus} \diff\mu(\alpha) \, \mathcal{H}_\alpha, \quad \dim(\mathcal{H}_\alpha) = 1.
\end{equation}
For each spectral parameter $\alpha \in \R$, the pure state $|\alpha\rangle$ is a simultaneous generalized eigenstate of all boundary operators:
\begin{equation}
  \Zop(\beta) |\alpha\rangle = Z_\alpha(\beta) |\alpha\rangle, \quad Z_\alpha(\beta) \in \R.
\end{equation}
Furthermore, for any $n$-boundary insertion $\{\beta_1, \dots, \beta_n\}$, the joint expectation value evaluated on $|\alpha\rangle$ factorizes identically:
\begin{equation}
  \langle \alpha | \prod_{i=1}^n \Zop(\beta_i) | \alpha \rangle \equiv \prod_{i=1}^n Z_\alpha(\beta_i),
  \label{eq:exact_factorization_proof}
\end{equation}
with identically vanishing connected correlators and zero statistical variance:
\begin{equation}
  \mathrm{Var}_\alpha(Z_1, \dots, Z_n) \equiv 0.
\end{equation}
\end{theorem}
\begin{proof}
Since $|\alpha\rangle$ is an eigenstate of each $\Zop(\beta_i)$ with scalar eigenvalue $Z_\alpha(\beta_i)$, the operator product acting on $|\alpha\rangle$ satisfies:
\begin{align}
  \langle \alpha | \Zop(\beta_1) \dots \Zop(\beta_n) | \alpha \rangle &= \langle \alpha | \left( \prod_{i=1}^n Z_\alpha(\beta_i) \right) | \alpha \rangle \nonumber\\
  &= \left( \prod_{i=1}^n Z_\alpha(\beta_i) \right) \langle \alpha | \alpha \rangle = \prod_{i=1}^n Z_\alpha(\beta_i).
\end{align}
The connected two-point correlator in an $\alpha$-state evaluates to:
\begin{equation}
  \langle \alpha | \Zop(\beta_1) \Zop(\beta_2) | \alpha \rangle - \langle \alpha | \Zop(\beta_1) | \alpha \rangle \langle \alpha | \Zop(\beta_2) | \alpha \rangle = Z_\alpha(\beta_1) Z_\alpha(\beta_2) - Z_\alpha(\beta_1) Z_\alpha(\beta_2) \equiv 0.
\end{equation}
The apparent non-factorization in the Hartle--Hawking state arises solely from averaging over the unobserved superselection parameter $\alpha$:
\begin{equation}
  \langle Z(\beta_1) Z(\beta_2) \rangle_{\mathrm{HH}} = \int_{\R} \diff\mu(\alpha) \, Z_\alpha(\beta_1) Z_\alpha(\beta_2) \neq \left( \int_{\R} \diff\mu(\alpha) Z_\alpha(\beta_1) \right) \left( \int_{\R} \diff\mu(\alpha) Z_\alpha(\beta_2) \right).
\end{equation}
Hence, factorization is exact in each pure physical state $|\alpha\rangle$, and the random matrix ensemble is simply a statistical mixture over ignorance of the true cosmological state.
\end{proof}

\begin{theorem}[Null-State Quotient Theorem]
\label{thm:null_state_quotient}
Let $\{|k\rangle\}_{k=0}^K$ denote the multi-universe state basis generated by acting with $k$ boundary operators on the no-boundary vacuum: $|k\rangle = (\Zop)^k \HH$. The topological inner product Gram matrix:
\begin{equation}
  G_{ij} = \langle k_i | k_j \rangle_{\Hbu} = \braHH (\Zop^\dagger)^i (\Zop)^j \HH

\end{equation}
is symmetric and positive semi-definite ($G \ge 0$). When the Fock truncation dimension $K+1$ exceeds the Bekenstein--Hawking microstate bound $N_{\max} = \lfloor \mathrm{e}^{S_0} \rfloor$, the Gram matrix develops a non-trivial kernel:
\begin{equation}
  \mathcal{N} = \left\{ |\psi\rangle \in \Hbu : \langle \psi | \psi \rangle = 0 \right\} \neq \{0\}.
\end{equation}
The physical baby universe Hilbert space is the quotient space~\cite{MarolfMaxfield2020,McNamaraVafa2020}:
\begin{equation}
  \Hphys = \Hbu / \mathcal{N}, \quad \mathrm{with} \quad \dim(\Hphys) = \mathrm{rank}(G) \le \lfloor \mathrm{e}^{S_0} \rfloor.
\end{equation}
\end{theorem}
\begin{proof}
The bulk inner product computed by the gravitational path integral is holographic dual to an ensemble expectation value over boundary Hamiltonians $H \in \mathrm{End}(\mathcal{H}_{\mathrm{boundary}})$ with finite microstate dimension $\dim(\mathcal{H}_{\mathrm{boundary}}) = d \le \lfloor \mathrm{e}^{S_0} \rfloor$. \begin{equation}
  p(M) = M^d + c_{d-1} M^{d-1} + \dots + c_0 \mathbb{I} \equiv 0.
\end{equation}
Taking the trace yields the fundamental trace identity $\Tr(M^d) + \sum_{j=0}^{d-1} c_j \Tr(M^j) = 0$. Under the holographic bulk-to-boundary homomorphism $\pi: \Hbu \to \mathbb{C}[\Tr(M^k)]$ where multi-boundary creation operators map to boundary trace polynomials, Cayley--Hamilton restricts the independent basis of invariant observables: any power $k \ge d$ reduces linearly modulo null vectors.

Consequently, the state vector $|\chi\rangle = p(\Zop) \HH \in \Hbu$ has vanishing topological norm:
\begin{equation}
  \langle \chi | \chi \rangle_{\Hbu} = \left\langle \HH \left| p(\Zop)^\dagger p(\Zop) \right| \HH \right\rangle = \overline{\left\langle p(M), p(M) \right\rangle}_{\mathrm{boundary}} \equiv 0.
\end{equation}
Therefore, any state with $k \ge d$ is a linear combination of states with $k < d$ modulo null vectors. Factoring out the kernel of all such zero-norm states defines the physical quotient $\Hphys = \Hbu / \mathcal{N}$, whose dimension is strictly bounded by $\dim(\Hphys) = \mathrm{rank}(G) \le d \le \lfloor \mathrm{e}^{S_0} \rfloor$.
\end{proof}

\begin{proof}

\begin{align}

\end{align}

\begin{equation}
  \int_{\mathcal{M}} \diff^2 x \sqrt{g} \, |\nabla \psi|^2 \ge \frac{\pi^2}{b^2} \int_{\mathcal{M}} \diff^2 x \sqrt{g} \, |\psi|^2.
\end{equation}
Therefore:
\begin{equation}
  \langle \psi, \mathcal{D}_2 \psi \rangle \ge \left( \frac{\pi^2}{b^2} + \min_{x} V''(\Phi(x)) \right) \|\psi\|^2 = \lambda_0 \|\psi\|^2 > 0.
\end{equation}
Since $\lambda_0 > 0$, no negative eigenvalue modes exist in the transverse fluctuation sector. Normalized against the free Laplacian solution $\psi_0(r) = r$ with identical initial conditions, the determinant ratio evaluated at $r = b$ evaluates to $\frac{\psi(b)}{\psi_0(b)} = \frac{\sinh(m b)}{m b} > 0$, establishing positivity.
\end{proof}

\section{Baby Universe Radiation Leakage and Island Regularization}
\label{sec:islands}

In semiclassical evaporating black holes, the Island Rule~\cite{AlmheiriHartmanMaldacena2020} computes the von Neumann entropy of Hawking radiation by extremizing the generalized entropy functional over disjoint spatial regions $\mathrm{Is}$ in the gravitational bulk:
\begin{equation}
  S(\mathrm{Rad}) = \min_{\mathrm{Is}} \operatorname{ext}_{\mathrm{Is}} \left[ \frac{\mathrm{Area}(\partial \mathrm{Is})}{4 G_N} + \frac{\Phi(\partial \mathrm{Is})}{4 G_N} + S_{\mathrm{semi-cl}}(\mathrm{Rad} \cup \mathrm{Is}) \right].
  \label{eq:island_rule}
\end{equation}

When baby universes nucleate from the black hole interior, Hawking quanta emitted into the interior can escape through baby universe wormhole throats into disconnected baby universes.

\begin{theorem}[Island Page Time Turnaround Under Baby Universe Emission]
\label{thm:page_curve_shift}
Let $\rho_{\mathrm{BU}} \in [0, 1]$ denote the normalized rate of baby universe nucleation per evaporation interval. Under non-zero baby universe density, the generalized entropy functional receives an entropy sink contribution $\Delta S_{\mathrm{leak}}(t) = 2.4 \rho_{\mathrm{BU}} (t / t_{\mathrm{evap}})^{1.4}$. The effective radiation entropy becomes:
\begin{equation}
